## Challenge Summary

The remote service sends:

1. An integer `n`, the number of nodes.
2. An `n x n` adjacency matrix of edge weights.
3. A prompt asking for the minimum tour distance.

This is the Traveling Salesman Problem (TSP): find the minimum-cost Hamiltonian cycle that visits every node exactly once and returns to the start.

## Key Observation

The phrase `minimum tour distance` is the giveaway. The matrix gives the travel cost between every pair of cities, so we need the exact shortest round trip through all nodes.

For `n = 20`, brute force is impossible, but the classic Held-Karp dynamic programming solution works:

- State: `dp[mask][j]`
- Meaning: minimum cost to start at node `0`, visit the subset `mask` of nodes `1..n-1`, and end at node `j`
- Transition:

```text
dp[mask][j] = min(dp[mask without j][k] + dist[k][j])
```

- Final answer:

```text
min(dp[all_visited][j] + dist[j][0])
```

This runs in `O(n^2 * 2^n)`, which is fine for `n = 20` in optimized native code.

## Solve Strategy

I first verified the challenge behavior by connecting with:

```bash
nc nc.umbccd.net 23456
```

The service printed a 20x20 matrix and asked for `Enter minimum tour distance:`.

I then wrote a small client that:

- Connects to the remote service
- Parses `n` and the matrix
- Solves the TSP exactly with Held-Karp DP
- Sends the result back immediately

## Solver

```cpp
#include <bits/stdc++.h>
#include <sys/socket.h>
#include <netdb.h>
#include <unistd.h>
using namespace std;

static int tsp_exact(const vector<vector<int>>& a) {
    int n = (int)a.size();
    int N = n - 1;
    int FULL = 1 << N;
    const int INF = 1000000000;
    vector<int> dp(FULL * N, INF);
    auto IDX = [N](int mask, int j) { return mask * N + j; };

    for (int j = 0; j < N; ++j) dp[IDX(1 << j, j)] = a[0][j + 1];

    for (int mask = 1; mask < FULL; ++mask) {
        if (__builtin_popcount((unsigned)mask) == 1) continue;
        for (int j = 0; j < N; ++j) if (mask & (1 << j)) {
            int pmask = mask ^ (1 << j);
            int best = INF;
            int base = pmask * N;
            for (int k = 0; k < N; ++k) if (pmask & (1 << k)) {
                int cand = dp[base + k] + a[k + 1][j + 1];
                if (cand < best) best = cand;
            }
            dp[IDX(mask, j)] = best;
        }
    }

    int ans = INF;
    int mask = FULL - 1;
    int base = mask * N;
    for (int j = 0; j < N; ++j) ans = min(ans, dp[base + j] + a[j + 1][0]);
    return ans;
}

int main() {
    addrinfo hints{}, *res;
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_STREAM;

    if (getaddrinfo("nc.umbccd.net", "23456", &hints, &res) != 0) return 1;
    int sock = socket(res->ai_family, res->ai_socktype, res->ai_protocol);
    if (sock < 0) return 1;
    if (connect(sock, res->ai_addr, res->ai_addrlen) != 0) return 1;
    freeaddrinfo(res);

    string buf;
    char tmp[65536];
    const string prompt = "Enter minimum tour distance:";

    while (true) {
        ssize_t r = recv(sock, tmp, sizeof(tmp), 0);
        if (r <= 0) break;
        buf.append(tmp, tmp + r);

        while (true) {
            size_t pos = buf.find(prompt);
            if (pos == string::npos) break;

            string before = buf.substr(0, pos);
            vector<string> lines;
            stringstream ss(before);
            string line;
            while (getline(ss, line)) {
                if (!line.empty() && line.back() == '\r') line.pop_back();
                if (!line.empty()) lines.push_back(line);
            }

            int start = -1, n = -1;
            for (int i = (int)lines.size() - 1; i >= 0; --i) {
                bool digits = !lines[i].empty();
                for (char c : lines[i]) digits = digits && isdigit((unsigned char)c);
                if (digits) {
                    start = i;
                    n = stoi(lines[i]);
                    break;
                }
            }

            if (start == -1 || (int)lines.size() < start + 1 + n) break;

            vector<vector<int>> a(n, vector<int>(n));
            for (int i = 0; i < n; ++i) {
                stringstream rs(lines[start + 1 + i]);
                for (int j = 0; j < n; ++j) rs >> a[i][j];
            }

            int ans = tsp_exact(a);
            string out = to_string(ans) + "\n";
            send(sock, out.c_str(), out.size(), 0);
            buf.erase(0, pos + prompt.size());
        }
    }

    close(sock);
    return 0;
}
```
